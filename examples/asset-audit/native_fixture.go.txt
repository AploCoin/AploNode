package main

import (
	"encoding/json"
	"github.com/ethereum/go-ethereum/common"
	"github.com/ethereum/go-ethereum/core"
	"github.com/ethereum/go-ethereum/core/types"
	"github.com/ethereum/go-ethereum/crypto"
	"github.com/ethereum/go-ethereum/params"
	"math/big"
	"os"
	"path/filepath"
)

func must(e error) {
	if e != nil {
		panic(e)
	}
}
func input(sig string, words ...[]byte) []byte {
	b := common.CopyBytes(crypto.Keccak256([]byte(sig))[:4])
	for _, w := range words {
		b = append(b, common.LeftPadBytes(w, 32)...)
	}
	return b
}
func slot(a common.Address) common.Hash {
	return crypto.Keccak256Hash(common.LeftPadBytes(a[:], 32), make([]byte, 32))
}
func main() {
	root := os.Args[1]
	must(os.MkdirAll(root, 0700))
	authorityKey, e := crypto.ToECDSA(common.LeftPadBytes(big.NewInt(1).Bytes(), 32))
	must(e)
	authority := crypto.PubkeyToAddress(authorityKey.PublicKey)
	config := *params.AllCliqueProtocolChanges
	config.ChainID = big.NewInt(424242)
	config.ChainID_ALT = big.NewInt(424242)
	config.Clique = &params.CliqueConfig{Period: 0, Epoch: 30000}
	alloc := core.GenesisAlloc{common.HexToAddress("0x1237"): {Nonce: 1, Balance: new(big.Int)}, params.GAploContractAddress: {Code: common.FromHex(params.GAPLO), Balance: new(big.Int), Storage: map[common.Hash]common.Hash{}}, params.AploContractAddress: {Code: []byte{0}, Balance: new(big.Int)}, params.BlockOracleContractAddress: {Code: []byte{0}, Balance: new(big.Int)}, common.HexToAddress("0xba1a"): {Code: common.FromHex("6000353160005260206000f3"), Balance: new(big.Int)}, common.HexToAddress("0xba1b"): {Code: common.FromHex("4760005260206000f3"), Balance: big.NewInt(77)}}
	type actor struct {
		Name    string         `json:"name"`
		Address common.Address `json:"address"`
		Native  string         `json:"aplo"`
		Gas     string         `json:"gaplo"`
		Raw     string         `json:"raw,omitempty"`
	}
	var actors []actor
	supply := new(big.Int)
	for i := int64(1); i <= 12; i++ {
		key, e := crypto.ToECDSA(common.LeftPadBytes(big.NewInt(i).Bytes(), 32))
		must(e)
		address := crypto.PubkeyToAddress(key.PublicKey)
		native := big.NewInt(123)
		gas := big.NewInt(500000)
		value := big.NewInt(7)
		target := common.HexToAddress("0xb00b")
		data := []byte(nil)
		limit := uint64(100000)
		name := "ordinary_transfer"
		switch i {
		case 1:
			name = "authority"
			native.Exp(big.NewInt(10), big.NewInt(24), nil)
			gas.Exp(big.NewInt(10), big.NewInt(24), nil)
		case 3:
			name = "aplo_only"
			gas.SetInt64(0)
		case 4:
			name = "gaplo_only_value"
			native.SetInt64(0)
		case 5:
			name = "exact_fee_plus_value"
			gas.SetInt64(200000)
		case 6:
			name = "exact_native_value"
			native.SetInt64(7)
		case 7:
			name = "stake"
			native.Mul(big.NewInt(1000), new(big.Int).Exp(big.NewInt(10), big.NewInt(18), nil))
			gas.SetInt64(10000000)
			target = params.AploContractAddress
			value.SetInt64(0)
			data = input("stake(uint256)", native.Bytes())
			limit = 150000
		case 8:
			name = "gaplo_only_no_value"
			native.SetInt64(0)
			value.SetInt64(0)
		case 9:
			name = "gaplo_transfer"
			value.SetInt64(0)
			target = params.GAploContractAddress
			data = input("transfer(address,uint256)", common.HexToAddress("0xb00b").Bytes(), big.NewInt(10).Bytes())
		case 10:
			name = "dynamic_value"
		case 11:
			name = "empty"
			native.SetInt64(0)
			gas.SetInt64(0)
		case 12:
			name = "zero_fee_value"
		}
		alloc[address] = core.GenesisAccount{Balance: native}
		alloc[params.GAploContractAddress].Storage[slot(address)] = common.BigToHash(gas)
		supply.Add(supply, gas)
		a := actor{Name: name, Address: address, Native: native.String(), Gas: gas.String()}
		if i != 1 && i != 11 {
			tx := types.NewTransaction(0, target, value, limit, big.NewInt(2), data)
			if i == 10 {
				tx = types.NewTx(&types.DynamicFeeTx{ChainID: config.ChainID, Nonce: 0, GasTipCap: big.NewInt(1), GasFeeCap: big.NewInt(2), Gas: limit, To: &target, Value: value})
			}
			if i == 12 {
				tx = types.NewTransaction(0, target, value, limit, new(big.Int), data)
			}
			signed, e := types.SignTx(tx, types.LatestSignerForChainID(config.ChainID), key)
			must(e)
			raw, e := signed.MarshalBinary()
			must(e)
			a.Raw = "0x" + common.Bytes2Hex(raw)
		}
		actors = append(actors, a)
	}
	alloc[params.GAploContractAddress].Storage[common.HexToHash("0x2")] = common.BigToHash(supply)
	gen := core.Genesis{Config: &config, Alloc: alloc, GasLimit: 30000000, Difficulty: big.NewInt(1), BaseFee: big.NewInt(1), ExtraData: append(append(make([]byte, 32), authority[:]...), make([]byte, 65)...)}
	write := func(name string, v interface{}) {
		b, e := json.MarshalIndent(v, "", "  ")
		must(e)
		must(os.WriteFile(filepath.Join(root, name), b, 0600))
	}
	write("genesis.json", gen)
	write("actors.json", actors)
	must(os.WriteFile(filepath.Join(root, "synthetic-authority.key"), []byte(common.Bytes2Hex(crypto.FromECDSA(authorityKey))), 0600))
	must(os.WriteFile(filepath.Join(root, "password.txt"), []byte("synthetic-disposable-only\n"), 0600))
}
